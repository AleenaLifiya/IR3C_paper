% \imgcube{x}{y}{w}{h}{depth}{prefix}: a volume whose three visible faces are slices
\newcommand{\imgcube}[6]{%
  \pgfmathsetmacro{\ox}{#5*\dx}\pgfmathsetmacro{\oy}{#5*\dy}%
  \node[anchor=south west,inner sep=0] at (#1,#2)
       {\includegraphics[width=#3cm,height=#4cm]{#6_front}};
  \begin{scope}[shift={($(#1,#2)+(#3,0)$)},cm={\dx,\dy,0,1,(0,0)}]
    \node[anchor=south west,inner sep=0,transform shape] at (0,0)
         {\includegraphics[width=#5cm,height=#4cm]{#6_side}};
    \fill[black,opacity=0.22] (0,0) rectangle (#5,#4);
  \end{scope}
  \begin{scope}[shift={($(#1,#2)+(0,#4)$)},cm={1,0,\dx,\dy,(0,0)}]
    \node[anchor=south west,inner sep=0,transform shape] at (0,0)
         {\includegraphics[width=#3cm,height=#5cm]{#6_top}};
    \fill[white,opacity=0.08] (0,0) rectangle (#3,#5);
  \end{scope}
  \draw[ink,line width=0.45pt,line join=round]
    (#1,#2) -- ++(#3,0) -- ++(\ox,\oy) -- ++(0,#4) -- ++(-#3,0) -- ++(-\ox,-\oy)
    -- cycle  (#1,#2+#4) -- ++(#3,0) -- ++(\ox,\oy)  (#1+#3,#2) -- ++(0,#4);
}

% \block{x}{y}{w}{h}{depth}{colour}: a feature-map cuboid (w ~ channels, h,depth ~ size)
\newcommand{\block}[6]{%
  \pgfmathsetmacro{\ox}{#5*\dx}\pgfmathsetmacro{\oy}{#5*\dy}%
  \fill[#6!62!white] (#1,#2) rectangle ++(#3,#4);
  \fill[#6!34!white] (#1,#2+#4) -- ++(#3,0) -- ++(\ox,\oy) -- ++(-#3,0) -- cycle;
  \fill[#6!82!black!92] (#1+#3,#2) -- ++(\ox,\oy) -- ++(0,#4) -- ++(-\ox,-\oy) -- cycle;
  \draw[#6!55!black,line width=0.35pt,line join=round]
    (#1,#2) -- ++(#3,0) -- ++(\ox,\oy) -- ++(0,#4) -- ++(-#3,0) -- ++(-\ox,-\oy) -- cycle
    (#1,#2+#4) -- ++(#3,0) -- ++(\ox,\oy)  (#1+#3,#2) -- ++(0,#4);
}
% skip copy: open (white) cuboid
\newcommand{\copyblock}[5]{%
  \pgfmathsetmacro{\ox}{#5*\dx}\pgfmathsetmacro{\oy}{#5*\dy}%
  \fill[white] (#1,#2) rectangle ++(#3,#4);
  \fill[white!93!black] (#1,#2+#4) -- ++(#3,0) -- ++(\ox,\oy) -- ++(-#3,0) -- cycle;
  \fill[white!84!black] (#1+#3,#2) -- ++(\ox,\oy) -- ++(0,#4) -- ++(-\ox,-\oy) -- cycle;
  \draw[muted,line width=0.35pt,line join=round]
    (#1,#2) -- ++(#3,0) -- ++(\ox,\oy) -- ++(0,#4) -- ++(-#3,0) -- ++(-\ox,-\oy) -- cycle
    (#1,#2+#4) -- ++(#3,0) -- ++(\ox,\oy)  (#1+#3,#2) -- ++(0,#4);
}
% flat textured card (for the 6-channel stack)
\newcommand{\card}[5]{%
  \node[anchor=south west,inner sep=0] at (#1,#2)
       {\includegraphics[width=#3cm,height=#4cm]{#5}};
  \draw[ink,line width=0.35pt] (#1,#2) rectangle ++(#3,#4);
}
% mini line plot of one positional profile
\newcommand{\profplot}[5]{% x y name colour coords
  \begin{scope}[shift={(#1,#2)},x=0.0640cm,y=0.36cm]
    \foreach \t in \profknownticks {\draw[rule,line width=0.3pt] (\t,0) -- (\t,1);}
    \draw[rule,line width=0.35pt] (0,0) -- (16,0);
    \draw[#4,line width=0.75pt,line join=round] plot coordinates {#5};
    \foreach \p in {#5} {\fill[#4] \p circle (0.55pt);}
    \node[sub,anchor=east,text=ink] at (-0.9,0.40) {#3};
  \end{scope}
}
